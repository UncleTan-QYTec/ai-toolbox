### 1. 前端配置（app.js）

const CONFIG = {
    // ✅ 后端 API 地址已配置
    backendApiUrl: 'http://localhost:3000/api/ppt',
    
    // 轮询间隔 3 秒

### 2. 生成调用（app.js）

```javascript
async function generatePPT() {
    // ✅ 调用后端 API
    const response = await fetch(`${CONFIG.backendApiUrl}/generate`, {
        method: 'POST',
        body: JSON.stringify({ topic, slideCount, style, language })
    });
    
    const data = await response.json();
    currentTaskId = data.taskId;
    
    // ✅ 开始轮询进度
    pollTaskStatus(data.taskId);
}
```

### 3. 后端接收（server.js）

```javascript
app.post('/api/ppt/generate', async (req, res) => {
    // ✅ 接收前端请求
    const { topic, slideCount, style, language } = req.body;
    
    // ✅ 创建任务
    const taskId = uuidv4();
    
    // ✅ 调用 Skywork 技能
    exec(`python3 scripts/run_ppt_write.py ...`);
    
    // ✅ 立即返回任务 ID
    res.json({ taskId, success: true });
});
```

### 4. 进度轮询（app.js → server.js）

```javascript
// 前端每 3 秒轮询
function pollTaskStatus(taskId) {
    setInterval(async () => {
        const response = await fetch(`${CONFIG.backendApiUrl}/status/${taskId}`);
        const task = (await response.json()).task;
        
        // ✅ 更新进度条
        updateProgress(task.progress, task.stage);
        
        // ✅ 检查是否完成
        if (task.status === 'completed') {
            showResult();
        }
    }, 3000);
}
```
